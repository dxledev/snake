cmake_minimum_required(VERSION 3.21)
project(Snake VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(SNAKE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SNAKE_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Widgets OpenGLWidgets)
add_library(snake_core src/core/game.cpp src/core/theme.cpp src/core/theme_source.cpp src/core/storage.cpp)
target_include_directories(snake_core PUBLIC src)
target_link_libraries(snake_core PUBLIC Qt6::Core Qt6::Gui)
add_library(snake_ui src/ui/board.cpp src/ui/icon_button.cpp src/ui/modal.cpp src/ui/modal_backdrop.cpp src/ui/modal_motion.cpp src/ui/window.cpp src/ui/theme_watcher.cpp)
target_link_libraries(snake_ui PUBLIC snake_core Qt6::Widgets Qt6::OpenGLWidgets)
add_executable(snake src/main.cpp src/cli.cpp)
target_link_libraries(snake PRIVATE snake_ui)
foreach(target snake_core snake_ui snake)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    foreach(name game theme ui)
        add_executable(${name}_tests tests/${name}_tests.cpp)
        if(name STREQUAL "game")
            target_link_libraries(${name}_tests PRIVATE snake_core Qt6::Test)
        else()
            target_link_libraries(${name}_tests PRIVATE snake_ui Qt6::Test)
        endif()
        add_test(NAME ${name} COMMAND ${name}_tests)
        set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endforeach()
    add_test(NAME install COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/install_test.sh ${CMAKE_CURRENT_SOURCE_DIR} $<TARGET_FILE_DIR:snake>)
    add_test(NAME cli COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.sh $<TARGET_FILE:snake>)
endif()
include(GNUInstallDirs)
install(TARGETS snake RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/io.github.snake.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/io.github.snake.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
